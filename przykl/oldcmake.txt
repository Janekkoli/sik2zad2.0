cmake_minimum_required(VERSION 3.20)
project(
        zad2_sik
        VERSION 1.0
        LANGUAGES CXX)

set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra -O2")
set(CMAKE_CXX_STANDARD 23)

add_library(Common src/common.h src/err.cpp src/err.h
        src/common.cpp
        src/Trick.cpp
        src/Trick.h
        src/Game.cpp
        src/Game.h
        src/Allgames.cpp
        src/Allgames.h
        src/Player.cpp
        src/Player.h
        src/Card.cpp
        src/Card.h
        src/common_f.cpp
        src/common_f.h)

add_executable(kierki-klient src/kierki-klient.cpp src/klient_f.h src/klient_f.cpp )

target_link_libraries(kierki-klient PRIVATE Common)

add_executable(kierki-serwer src/kierki-serwer.cpp)

target_link_libraries(kierki-serwer PRIVATE Common)

add_executable(main src/main.cpp)

target_link_libraries(main PRIVATE Common)
